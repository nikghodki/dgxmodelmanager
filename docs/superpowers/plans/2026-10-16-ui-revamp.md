# DGXMM UI Revamp Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the DGXMM frontend's home-made components with a professional design system (shadcn/ui on Radix + Tailwind tokens). The result has light (Enterprise) and dark (Graphite) themes, a grouped sidebar plus top bar, a ⌘K command palette, mobile bottom tabs, responsive data tables, toasts, and restyled, accessible charts. Every page keeps its current behaviour and API calls.

**Architecture:**
- Only `frontend/` changes.
- A token layer (CSS variables in `src/index.css`, mapped in `tailwind.config.js`) feeds shadcn primitives in `src/components/ui/` (generated by the shadcn CLI).
- App components in `src/components/app/` sit on top of the primitives.
- Pure logic that decides what to show lives in `src/lib/*.ts`, with unit tests.
- Pages in `src/pages/` are rewritten on the new components. They keep their react-query keys, SSE wiring (`stream.ts`), `api.ts` and `auth.tsx` unchanged.

**Tech Stack:**
- Runtime: React 18, TypeScript 5 strict, Vite 5, Tailwind 3.4 + `tailwindcss-animate`, shadcn/ui (CLI `shadcn@2.3.0`, Tailwind-v3 compatible, style "new-york"), Radix UI, lucide-react, cmdk, sonner, @tanstack/react-table 8, @tanstack/react-query 5, Recharts 2, react-router 6.
- Testing: vitest 2 + @testing-library/react 16 + jsdom.

**Spec:** `docs/superpowers/specs/2026-10-16-ui-revamp-design.md` (same repo). Read it before starting.

## Review Focus

The five failure modes most likely to bite a real user that the spec implies but no feature test would naturally cover. Each is pinned by a test in the owning task.

1. **Viewer role sees admin-only actions.** Expected: viewers see no Deploy, Stop/Sleep/Wake/Remove, Users, Audit, adopt or download actions, in the sidebar, row menus, command palette or headers. → tests in Task 7 (`visibleGroups`), Task 8 (`buildCommands`), Task 10 (`deploymentActions`).
2. **Missing or non-numeric live data** (no SSE samples yet, NaN/Infinity, deployments with no metrics). Expected: tiles and cells show "—", never `NaN`, and nothing throws. → tests in Task 10 (`fleetKpis`) and Task 6 (`MetricChart` empty state).
3. **Very long identifiers on a phone** (60+ char repo ids, absolute GGUF paths, long error text). Expected: text truncates or wraps inside its cell or card with the full value in a tooltip/`title`, and the page never scrolls sideways. → tests in Task 5 (`DataTable` mobile card wraps) and Task 10 (model cell `title`).
4. **Server errors on actions** (`insufficient_memory`, `rollback_incomplete`, 403). Expected: a toast shows the server's `detail.message`, and buttons re-enable. → test in Task 4 (`useActionMutation`).
5. **Theme persistence and OS changes.** Expected: the chosen theme survives reloads (no flash), and in `system` mode an OS light/dark switch applies live. → tests in Task 3.

```tsx
import { createContext, useCallback, useContext, useEffect, useState, type ReactNode } from "react";
import { Laptop, Moon, Sun } from "lucide-react";
import { Button } from "@/components/ui/button";
import {
  DropdownMenu, DropdownMenuContent, DropdownMenuRadioGroup, DropdownMenuRadioItem, DropdownMenuTrigger,
} from "@/components/ui/dropdown-menu";

export type Theme = "light" | "dark" | "system";
export const THEME_KEY = "dgxmm-theme";

const systemDark = () => window.matchMedia?.("(prefers-color-scheme: dark)").matches ?? false;

export function applyTheme(t: Theme): "light" | "dark" {
  const resolved = t === "system" ? (systemDark() ? "dark" : "light") : t;
  document.documentElement.classList.toggle("dark", resolved === "dark");
  document.documentElement.style.colorScheme = resolved;
  return resolved;
}

function readSaved(): Theme {
  const v = (typeof localStorage !== "undefined" && localStorage.getItem(THEME_KEY)) || "system";
  return v === "light" || v === "dark" || v === "system" ? v : "system";
}

interface ThemeState { theme: Theme; resolved: "light" | "dark"; setTheme: (t: Theme) => void }
const Ctx = createContext<ThemeState>({ theme: "system", resolved: "light", setTheme: () => undefined });

export function ThemeProvider({ children }: { children: ReactNode }) {
  const [theme, setThemeState] = useState<Theme>(readSaved);
  const [resolved, setResolved] = useState<"light" | "dark">(() => applyTheme(readSaved()));

  useEffect(() => {
    setResolved(applyTheme(theme));
    if (theme !== "system") return;
    const mq = window.matchMedia("(prefers-color-scheme: dark)");
    const onChange = () => setResolved(applyTheme("system"));
    mq.addEventListener("change", onChange);
    return () => mq.removeEventListener("change", onChange);
  }, [theme]);

  const setTheme = useCallback((t: Theme) => {
    localStorage.setItem(THEME_KEY, t);
    setThemeState(t);
  }, []);

  return <Ctx.Provider value={{ theme, resolved, setTheme }}>{children}</Ctx.Provider>;
}

export const useTheme = () => useContext(Ctx);

export function ThemeToggle() {
  const { theme, resolved, setTheme } = useTheme();
  return (
    <DropdownMenu>
      <DropdownMenuTrigger asChild>
        <Button variant="ghost" size="icon" aria-label="Change theme">
          {resolved === "dark" ? <Moon className="h-4 w-4" /> : <Sun className="h-4 w-4" />}
        </Button>
      </DropdownMenuTrigger>
      <DropdownMenuContent align="end">
        <DropdownMenuRadioGroup value={theme} onValueChange={(v) => setTheme(v as Theme)}>
          <DropdownMenuRadioItem value="light"><Sun className="mr-2 h-4 w-4" />Light</DropdownMenuRadioItem>
          <DropdownMenuRadioItem value="dark"><Moon className="mr-2 h-4 w-4" />Dark</DropdownMenuRadioItem>

        </DropdownMenuRadioGroup>
      </DropdownMenuContent>
```

```ts
import type { Budget } from "@/types";

export interface MemorySegment {
  key: string;
  label: string;
  bytes: number;
  pct: number;
  color: string;
  kind: "deployment" | "sleeping" | "external" | "reserve" | "free";
}

const ACTIVE = ["starting", "loading", "ready", "sleeping", "unknown", "stopping"];

/** Segments for the unified-memory bar. Deployment colour = chart slot by deployment id (follows the entity). */
export function memorySegments(b: Budget): { segments: MemorySegment[]; overcommitted: number } {
  const total = b.total || 1;
  const deps = b.deployments.filter((d) => ACTIVE.includes(d.state)).sort((x, y) => x.deployment_id - y.deployment_id);
  const segs: Omit<MemorySegment, "pct">[] = deps.map((d) => {
    const sleeping = d.state === "sleeping";
    return {
      key: `dep-${d.deployment_id}`,
      label: sleeping ? `${d.name} · sleeping` : d.name,
      bytes: sleeping ? d.actual : Math.max(d.reserved, d.actual),
      color: `var(--chart-${((d.deployment_id - 1) % 8 + 8) % 8 + 1})`,
      kind: sleeping ? "sleeping" : "deployment",
    };
```

```ts
import { useMutation, useQueryClient } from "@tanstack/react-query";
import { toast } from "sonner";

interface Options<TData, TVars> {
  success?: string | ((data: TData, vars: TVars) => string);
  invalidate?: unknown[][];
  onSuccess?: (data: TData, vars: TVars) => void;
}

/** A react-query mutation that reports its outcome as a toast and refreshes the given queries. */
export function useActionMutation<TVars = void, TData = unknown>(fn: (vars: TVars) => Promise<TData>, opts: Options<TData, TVars> = {}) {
  const qc = useQueryClient();
  return useMutation<TData, Error, TVars>({
    mutationFn: fn,
    onSuccess: (data, vars) => {
      if (opts.success) toast.success(typeof opts.success === "function" ? opts.success(data, vars) : opts.success);
      opts.onSuccess?.(data, vars);
    },
    onError: (err) => toast.error(err.message || "Request failed"),
    onSettled: () => {
      for (const key of opts.invalidate ?? []) qc.invalidateQueries({ queryKey: key });
    },
  });
}
```

```tsx
/** Minimal 2px line, no axes. `values` oldest→newest; nulls are gaps. */
export function Sparkline({ values, label, color = "var(--chart-1)" }: { values: (number | null)[]; label: string; color?: string }) {
  const nums = values.filter((v): v is number => v != null && Number.isFinite(v));
  if (nums.length < 2) return <div className="h-6" aria-hidden />;
  const max = Math.max(...nums), min = Math.min(...nums), span = max - min || 1;
  const step = 100 / (values.length - 1);
  const parts: string[] = [];
  values.forEach((v, i) => {
    if (v == null || !Number.isFinite(v)) return;
    parts.push(`${(i * step).toFixed(2)},${(22 - ((v - min) / span) * 20).toFixed(2)}`);
  });
  return (
    <svg viewBox="0 0 100 24" preserveAspectRatio="none" className="h-6 w-full" role="img" aria-label={label}>
      <title>{label}</title>
      <polyline fill="none" stroke={color} strokeWidth={2} vectorEffect="non-scaling-stroke" strokeLinejoin="round" points={parts.join(" ")} />
    </svg>
  );
}
```

```ts
import { useEffect, useState } from "react";

export function useMediaQuery(query: string): boolean {
  const get = () => (typeof window !== "undefined" && window.matchMedia ? window.matchMedia(query).matches : false);
  const [matches, setMatches] = useState(get);
  useEffect(() => {
    const mq = window.matchMedia(query);
    const on = () => setMatches(mq.matches);
    on();
    mq.addEventListener?.("change", on);
    return () => mq.removeEventListener?.("change", on);
  }, [query]);
  return matches;
}

export const useIsMobile = () => useMediaQuery("(max-width: 767px)");
```

```tsx
export function MetricChart({ data, series, height = 200, fmt = defaultFmt, xKey = "ts", xFmt = timeTick }: {

  if (!data.length) {
    return <div className="flex items-center justify-center rounded-md border border-dashed text-xs text-muted-foreground" style={{ height }}>No data yet</div>;
  }
  const clean = finiteSeries(data, series.map((s) => s.key));
  return (

          contentStyle={{ background: c.tooltipBg, border: `1px solid ${c.tooltipBorder}`, borderRadius: 8, fontSize: 12, color: c.text }}
          labelFormatter={(v) => (xKey === "ts" ? timeFull(v as number) : String(v))}
```

```ts
import type { Theme } from "./theme";
import { visibleGroups } from "./nav";

export type CommandAction =
  | { type: "navigate"; to: string }
  | { type: "deployment"; op: "sleep" | "wake" | "stop"; id: number; name: string }
  | { type: "theme"; theme: Theme }
  | { type: "logout" };

export interface CommandDef {
  id: string;
  group: "Pages" | "Deployments" | "Models" | "Preferences";
  label: string;
  keywords: string;
  action: CommandAction;
  destructive?: boolean;
}

interface Dep { id: number; name: string; state: string; sleep_capable: boolean; model_ref?: string }
interface Art { ref: string; complete: boolean; kind: string; engines: string[] }

const STOPPABLE = ["ready", "sleeping", "starting", "loading", "unknown"];

export function buildCommands({ isAdmin, deployments, artifacts }: { isAdmin: boolean; deployments: Dep[]; artifacts: Art[] }): CommandDef[] {
  const out: CommandDef[] = [];
  for (const g of visibleGroups(isAdmin)) for (const i of g.items) {
    out.push({ id: `page:${i.to}`, group: "Pages", label: i.label, keywords: `${g.label} page go`, action: { type: "navigate", to: i.to } });
  }
  if (isAdmin) out.push({ id: "page:/deploy", group: "Pages", label: "Deploy a model", keywords: "new create launch", action: { type: "navigate", to: "/deploy" } });
  out.push({ id: "page:/change-password", group: "Pages", label: "Change password", keywords: "account", action: { type: "navigate", to: "/change-password" } });

  for (const d of deployments) {
    out.push({ id: `dep:${d.id}:open`, group: "Deployments", label: `Open ${d.name}`, keywords: `${d.model_ref ?? ""} ${d.state}`, action: { type: "navigate", to: `/deployments/${d.id}` } });
    if (!isAdmin) continue;
    if (d.state === "ready" && d.sleep_capable) out.push({ id: `dep:${d.id}:sleep`, group: "Deployments", label: `Sleep ${d.name}`, keywords: "free memory", action: { type: "deployment", op: "sleep", id: d.id, name: d.name } });
    if (d.state === "sleeping") out.push({ id: `dep:${d.id}:wake`, group: "Deployments", label: `Wake ${d.name}`, keywords: "resume", action: { type: "deployment", op: "wake", id: d.id, name: d.name } });
    if (STOPPABLE.includes(d.state)) out.push({ id: `dep:${d.id}:stop`, group: "Deployments", label: `Stop ${d.name}`, keywords: "shutdown kill", action: { type: "deployment", op: "stop", id: d.id, name: d.name }, destructive: true });
  }

  if (isAdmin) for (const a of artifacts) {
    if (!a.complete || a.kind !== "llm" || !a.engines.length) continue;
    out.push({ id: `model:${a.ref}`, group: "Models", label: `Deploy ${a.ref}`, keywords: a.engines.join(" "), action: { type: "navigate", to: `/deploy?model=${encodeURIComponent(a.ref)}` } });
  }

  for (const t of ["light", "dark", "system"] as const) {
    out.push({ id: `theme:${t}`, group: "Preferences", label: `Theme: ${t[0].toUpperCase()}${t.slice(1)}`, keywords: "appearance mode", action: { type: "theme", theme: t } });
  }
  out.push({ id: "logout", group: "Preferences", label: "Log out", keywords: "sign out", action: { type: "logout" } });
  return out;
}
```

```tsx
import { useMemo, useState } from "react";
import { useQuery } from "@tanstack/react-query";
import { useNavigate } from "react-router-dom";
import { ArrowRight, Box, LogOut, Moon, Palette, Power, Server } from "lucide-react";
import { api } from "@/api";
import { useAuth, useIsAdmin } from "@/auth";
import { CommandDialog, CommandEmpty, CommandGroup, CommandInput, CommandItem, CommandList } from "@/components/ui/command";
import { useActionMutation } from "@/hooks/use-action-mutation";
import type { Artifact, Deployment } from "@/types";
import { buildCommands, type CommandDef } from "./commands";
import { ConfirmDialog } from "./confirm-dialog";
import { useTheme } from "./theme";

const GROUPS: CommandDef["group"][] = ["Pages", "Deployments", "Models", "Preferences"];

export function CommandPalette({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
  const isAdmin = useIsAdmin();
  const { logout } = useAuth();
  const { setTheme } = useTheme();
  const nav = useNavigate();
  const deps = useQuery({ queryKey: ["deployments"], queryFn: () => api<Deployment[]>("/deployments"), enabled: open });
  const inv = useQuery({ queryKey: ["inventory"], queryFn: () => api<Artifact[]>("/inventory"), enabled: open && isAdmin });
  const [confirmStop, setConfirmStop] = useState<{ id: number; name: string } | null>(null);
  const op = useActionMutation(({ id, op }: { id: number; op: string }) => api(`/deployments/${id}/${op}`, { method: "POST" }), {
    success: (_d, v) => `${v.op[0].toUpperCase()}${v.op.slice(1)} requested`,
    invalidate: [["deployments"]],
  });
  const commands = useMemo(
    () => buildCommands({ isAdmin, deployments: deps.data ?? [], artifacts: inv.data ?? [] }),
    [isAdmin, deps.data, inv.data],
  );

  const run = (c: CommandDef) => {
    onOpenChange(false);
    const a = c.action;
    if (a.type === "navigate") nav(a.to);
    else if (a.type === "theme") setTheme(a.theme);
    else if (a.type === "logout") logout();
    else if (a.op === "stop") setConfirmStop({ id: a.id, name: a.name });
    else op.mutate({ id: a.id, op: a.op });
  };

  const icon = (c: CommandDef) => {
    if (c.action.type === "theme") return <Palette className="mr-2 h-4 w-4" />;
    if (c.action.type === "logout") return <LogOut className="mr-2 h-4 w-4" />;
    if (c.action.type === "deployment") return c.action.op === "stop" ? <Power className="mr-2 h-4 w-4" /> : <Moon className="mr-2 h-4 w-4" />;
    if (c.group === "Deployments") return <Server className="mr-2 h-4 w-4" />;
    if (c.group === "Models") return <Box className="mr-2 h-4 w-4" />;
    return <ArrowRight className="mr-2 h-4 w-4" />;
  };

  return (
    <>
      <CommandDialog open={open} onOpenChange={onOpenChange}>
        <CommandInput placeholder="Search pages, deployments, models, actions…" />
        <CommandList>
          <CommandEmpty>No matches.</CommandEmpty>
          {GROUPS.map((g) => {
            const items = commands.filter((c) => c.group === g);
            if (!items.length) return null;
            return (
              <CommandGroup key={g} heading={g}>
                {items.map((c) => (
                  <CommandItem key={c.id} value={`${c.label} ${c.keywords}`} onSelect={() => run(c)}
                    className={c.destructive ? "text-status-critical aria-selected:text-status-critical" : undefined}>
                    {icon(c)}{c.label}
                  </CommandItem>
                ))}
              </CommandGroup>
            );
          })}
        </CommandList>
      </CommandDialog>
      <ConfirmDialog open={!!confirmStop} onOpenChange={(o) => !o && setConfirmStop(null)}
        title={`Stop ${confirmStop?.name}?`} description="The container is stopped and its memory released. You can start it again later."
        confirmLabel="Stop" destructive pending={op.isPending}
        onConfirm={() => { if (confirmStop) op.mutate({ id: confirmStop.id, op: "stop" }, { onSettled: () => setConfirmStop(null) }); }} />
    </>
  );
}
```

```ts
export type DeploymentAction = "open" | "sleep" | "wake" | "switch" | "reconfigure" | "stop" | "start" | "remove";

/** Which row/header actions a user may take on a deployment in its current state. */
export function deploymentActions(d: { state: string; sleep_capable: boolean }, isAdmin: boolean): DeploymentAction[] {
  const out: DeploymentAction[] = ["open"];
  if (!isAdmin) return out;
  if (d.state === "ready" && d.sleep_capable) out.push("sleep");
  if (d.state === "sleeping") out.push("wake");
  if (d.state === "ready" || d.state === "sleeping") out.push("switch");
  if (["ready", "sleeping", "stopped", "failed"].includes(d.state)) out.push("reconfigure");
  if (["ready", "sleeping", "starting", "loading", "unknown"].includes(d.state)) out.push("stop");
  if (d.state === "stopped" || d.state === "failed") out.push("start");
  if (canRemove(d.state)) out.push("remove");
  return out;
}
```

```ts
export interface FleetKpis {
  genTps: number | null; genSeries: (number | null)[];
  ttftP50: number | null; ttftSeries: (number | null)[];
  kv: number | null; waiting: number | null; running: number | null;
}

const num = (v: unknown): number | null => (typeof v === "number" && Number.isFinite(v) ? v : null);
const agg = (vals: (number | null)[], how: "sum" | "max") => {
  const xs = vals.filter((v): v is number => v != null);
  if (!xs.length) return null;
  return how === "sum" ? xs.reduce((a, b) => a + b, 0) : Math.max(...xs);
};

/** Aggregate live samples (one array per ready deployment, oldest→newest) into dashboard KPIs. */
export function fleetKpis(samples: any[][], points = 30): FleetKpis {
  const lasts = samples.map((s) => s[s.length - 1]).filter(Boolean);
  const pick = (k: string) => lasts.map((x) => num(x[k]));
  const series = (k: string, how: "sum" | "max") => Array.from({ length: points }, (_, i) =>
    agg(samples.map((s) => num(s[s.length - points + i]?.[k])), how));
  return {
    genTps: agg(pick("gen_tps"), "sum"), genSeries: series("gen_tps", "sum"),
    ttftP50: agg(pick("ttft_p50"), "max"), ttftSeries: series("ttft_p50", "max"),
    kv: agg(pick("kv_usage"), "max"), waiting: agg(pick("waiting"), "sum"), running: agg(pick("running"), "sum"),
  };
}
```

```ts
export interface EstimatePart { key: string; label: string; bytes: number }

const ORDER: [keyof Estimate, string][] = [
  ["weights", "Weights"], ["draft_weights", "Draft weights"], ["kv_cache", "KV cache"], ["state", "Recurrent state"],
  ["activations", "Activations"], ["runtime", "Runtime (CUDA, graphs)"], ["margin", "Margin / calibration"],
];

export function estimateParts(e: Partial<Estimate>): EstimatePart[] {
  return ORDER.map(([key, label]) => ({ key, label, bytes: Number(e[key] ?? 0) })).filter((p) => p.bytes !== 0);
}
```

```ts
interface Row { config_hash: string; concurrency: number; tps_total: number; current?: boolean; ts: string }

export function benchmarkSeries(rows: Row[]) {
  const byHash = new Map<string, { current: boolean; latest: string }>();
  for (const r of rows) {
    const prev = byHash.get(r.config_hash);
    byHash.set(r.config_hash, { current: !!(prev?.current || r.current), latest: prev && prev.latest > r.ts ? prev.latest : r.ts });
  }
  const keys = [...byHash.entries()]
    .sort((a, b) => Number(b[1].current) - Number(a[1].current) || b[1].latest.localeCompare(a[1].latest))
    .map(([hash, v]) => ({ hash, label: `${hash.slice(0, 8)}${v.current ? " (current)" : ""}`, current: v.current }));
  const levels = [...new Set(rows.map((r) => r.concurrency))].sort((a, b) => a - b);
  const data = levels.map((c) => {
    const row: Record<string, number> = { concurrency: c };
    for (const k of keys) {
      const hit = rows.filter((r) => r.config_hash === k.hash && r.concurrency === c).sort((a, b) => b.ts.localeCompare(a.ts))[0];
      if (hit) row[k.hash] = hit.tps_total;
    }
    return row;
  });
  return { data, keys };
}
```

```ts
import type { PlanAction, Preview } from "./types";

export function submitState(preview: Preview | null, actions: PlanAction[], pending: boolean, mode: "deploy" | "switch" | "reconfigure") {
  if (pending) return { disabled: true, label: mode === "deploy" ? "Deploying…" : "Working… this can take minutes" };
  const base = mode === "deploy" ? "Deploy" : mode === "switch" ? "Switch" : "Apply and restart";
  if (!preview) return { disabled: true, label: base };
  if (!preview.fit.fits && actions.length === 0) return { disabled: true, label: "Choose a plan to fit" };
  if (actions.length) return { disabled: false, label: mode === "deploy" ? "Apply plan and deploy" : `Apply plan and ${base.toLowerCase()}` };
  return { disabled: false, label: base };
}
```

```ts
export interface Rate { bytes: number; t: number; bps: number | null }

const ALPHA = 0.3;

export function nextRate(prev: Rate | undefined, bytes: number, t: number): Rate {
  if (!prev) return { bytes, t, bps: null };
  const dt = t - prev.t;
  if (dt <= 0) return prev;
  const inst = Math.max(bytes - prev.bytes, 0) / dt;
  return { bytes, t, bps: prev.bps == null ? inst : ALPHA * inst + (1 - ALPHA) * prev.bps };
}

export function etaSeconds(r: Rate | undefined, remaining: number): number | null {
  if (!r || !r.bps || r.bps <= 0) return null;
  return Math.round(remaining / r.bps);
}

export function fmtDuration(s: number | null): string {
  if (s == null || !Number.isFinite(s)) return "—";
  if (s < 60) return `${Math.round(s)}s`;
  if (s < 3600) return `${Math.floor(s / 60)}m ${Math.round(s % 60)}s`;
  return `${Math.floor(s / 3600)}h ${Math.floor((s % 3600) / 60)}m`;
}
```

```ts
export function fmtRelative(iso: string, now: number = Date.now()): string {
  const s = Math.max(0, (now - Date.parse(iso)) / 1000);
  if (s < 60) return "just now";
  if (s < 3600) return `${Math.floor(s / 60)} min ago`;
  if (s < 86400) return `${Math.floor(s / 3600)} h ago`;
  return `${Math.floor(s / 86400)} d ago`;
}
```
